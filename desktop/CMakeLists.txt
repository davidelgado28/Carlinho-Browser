cmake_minimum_required(VERSION 3.24)
project(CarlinhoBrowser LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 6.11 REQUIRED COMPONENTS Core Gui Widgets WebEngineWidgets)
add_executable(carlinho-browser main.cpp browser_window.cpp browser_window.h ../core/privacy_engine.cpp ../core/privacy_engine.h)
target_include_directories(carlinho-browser PRIVATE ../core)
target_link_libraries(carlinho-browser PRIVATE Qt6::Core Qt6::Gui Qt6::Widgets Qt6::WebEngineWidgets)
if(APPLE)
  set_target_properties(carlinho-browser PROPERTIES OSX_ARCHITECTURES "x86_64;arm64" MACOSX_BUNDLE TRUE)
endif()
if(WIN32)
  set_target_properties(carlinho-browser PROPERTIES WIN32_EXECUTABLE TRUE)
endif()
